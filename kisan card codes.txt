/* AUTO MARATHI TRANSLATION */
async function translateName(text){
  name_en.innerText = text;

  if(text.length > 0){
    let res = await fetch(
      "https://translate.googleapis.com/translate_a/single?client=gtx&sl=en&tl=mr&dt=t&q=" 
      + encodeURIComponent(text)
    );

    let data = await res.json();
    name_mr.innerText = data[0][0][0];
  }else{
    name_mr.innerText = "";
  }
}

async function translateFather(text){

      + encodeURIComponent(text)
    );

  }else{
    father_mr.innerText = "";
  }
}
